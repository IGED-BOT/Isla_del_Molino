VERTEX SHADER CON NORMALES Y UV (PASO 9/10/12)
--------------------------------------------------------------------------------
  #version 300 es
  in vec3 aPosition;
  in vec3 aNormal;
  in vec2 aTexCoord;

  uniform mat4 uModelMatrix;
  uniform mat4 uViewMatrix;
  uniform mat4 uProjectionMatrix;

  out vec3 vNormal;
  out vec3 vWorldPosition;
  out vec2 vTexCoord;

  void main() {
      vec4 worldPosition = uModelMatrix * vec4(aPosition, 1.0);
      gl_Position = uProjectionMatrix * uViewMatrix * worldPosition;
      vWorldPosition = worldPosition.xyz;
      vNormal = normalize(mat3(uModelMatrix) * aNormal);
      vTexCoord = aTexCoord;
  }

FRAGMENT SHADER CON ILUMINACIÓN PUNTUAL (PASO 12)
--------------------------------------------------------------------------------
  #version 300 es
  precision highp float;

  in vec3 vNormal;
  in vec3 vWorldPosition;
  in vec2 vTexCoord;

  uniform sampler2D uTexture;
  uniform vec3 uLightPosition;
  uniform bool uEmissive;

  out vec4 outColor;

  void main() {
      vec3 colorTextura = texture(uTexture, vTexCoord).rgb;

      if (uEmissive) {
          outColor = vec4(colorTextura, 1.0);
          return;
      }

      vec3 N = normalize(vNormal);
      vec3 L = normalize(uLightPosition - vWorldPosition);

      float difusa = max(dot(N, L), 0.0);
      float ambiente = 0.08;

      vec3 colorFinal = colorTextura * (ambiente + difusa);
      outColor = vec4(colorFinal, 1.0);
  }

COMPILACIÓN Y ENLACE
--------------------------------------------------------------------------------
  function crearShader(gl, tipo, codigoFuente) {
      const shader = gl.createShader(tipo);
      gl.shaderSource(shader, codigoFuente);
      gl.compileShader(shader);

      if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
          const error = gl.getShaderInfoLog(shader);
          gl.deleteShader(shader);
          throw new Error("Error al compilar shader:\n" + error);
      }
      return shader;
  }

  const vertexShader = crearShader(gl, gl.VERTEX_SHADER, vertexShaderSource);
  const fragmentShader = crearShader(gl, gl.FRAGMENT_SHADER, fragmentShaderSource);

  const program = gl.createProgram();
  gl.attachShader(program, vertexShader);
  gl.attachShader(program, fragmentShader);
  gl.linkProgram(program);

  if (!gl.getProgramParameter(program, gl.LINK_STATUS)) {
      throw new Error("Error al enlazar programa:\n" + gl.getProgramInfoLog(program));
  }

VECTORES 3D (FUNCIONES PROPIAS)
--------------------------------------------------------------------------------
  function restarVec3(a, b) {
      return [a[0]-b[0], a[1]-b[1], a[2]-b[2]];
  }
  function longitudVec3(v) {
      return Math.hypot(v[0], v[1], v[2]);
  }
  function normalizarVec3(v) {
      const l = longitudVec3(v);
      return [v[0]/l, v[1]/l, v[2]/l];
  }
  function productoCruz(a, b) {
      return [
          a[1]*b[2] - a[2]*b[1],
          a[2]*b[0] - a[0]*b[2],
          a[0]*b[1] - a[1]*b[0]
      ];
  }

MATRICES 3×3 (PASO 3)
--------------------------------------------------------------------------------
  function matrizIdentidad() {
      return new Float32Array([1,0,0, 0,1,0, 0,0,1]);
  }
  function matrizTraslacion(tx, ty) {
      return new Float32Array([1,0,0, 0,1,0, tx,ty,1]);
  }
  function matrizRotacion(anguloRadianes) {
      const c = Math.cos(anguloRadianes);
      const s = Math.sin(anguloRadianes);
      return new Float32Array([c,s,0, -s,c,0, 0,0,1]);
  }
  function matrizEscala(sx, sy) {
      return new Float32Array([sx,0,0, 0,sy,0, 0,0,1]);
  }
  function multiplicarMat3(a, b) {
      const resultado = new Float32Array(9);
      for (let columna = 0; columna < 3; columna++) {
          for (let fila = 0; fila < 3; fila++) {
              let suma = 0;
              for (let k = 0; k < 3; k++) {
                  suma += a[k*3 + fila] * b[columna*3 + k];
              }
              resultado[columna*3 + fila] = suma;
          }
      }
      return resultado;
  }

  Composición: M = T · R · S  (S primero, luego R, luego T)

MATRIZ DE PERSPECTIVA (PASO 6)
--------------------------------------------------------------------------------
  function matrizPerspectiva(fovRadianes, aspect, near, far) {
      const f = 1.0 / Math.tan(fovRadianes / 2);
      const nf = 1 / (near - far);

      return new Float32Array([
          f/aspect, 0, 0, 0,
          0, f, 0, 0,
          0, 0, (far+near)*nf, -1,
          0, 0, (2*far*near)*nf, 0
      ]);
  }

MATRIZ LOOKAT (PASO 6)
--------------------------------------------------------------------------------
  function matrizLookAt(eye, target, up) {
      const zAxis = normalizarVec3(restarVec3(eye, target));
      const xAxis = normalizarVec3(productoCruz(up, zAxis));
      const yAxis = productoCruz(zAxis, xAxis);

      return new Float32Array([
          xAxis[0], yAxis[0], zAxis[0], 0,
          xAxis[1], yAxis[1], zAxis[1], 0,
          xAxis[2], yAxis[2], zAxis[2], 0,
          -(xAxis[0]*eye[0] + xAxis[1]*eye[1] + xAxis[2]*eye[2]),
          -(yAxis[0]*eye[0] + yAxis[1]*eye[1] + yAxis[2]*eye[2]),
          -(zAxis[0]*eye[0] + zAxis[1]*eye[1] + zAxis[2]*eye[2]),
          1
      ]);
  }

ESFERA POR LATITUD/LONGITUD (PASO 7)
--------------------------------------------------------------------------------
  ECUACIONES:
    x = r · sin(φ) · cos(θ)
    y = r · cos(φ)
    z = r · sin(φ) · sin(θ)
  donde φ = latitud (0 a π), θ = longitud (0 a 2π).

  function crearEsfera(radio, segmentosLatitud, segmentosLongitud) {
      const vertices = [];
      const indices = [];

      for (let latitud = 0; latitud <= segmentosLatitud; latitud++) {
          const v = latitud / segmentosLatitud;
          const phi = v * Math.PI;

          for (let longitud = 0; longitud <= segmentosLongitud; longitud++) {
              const u = longitud / segmentosLongitud;
              const theta = u * Math.PI * 2;

              const nx = Math.sin(phi) * Math.cos(theta);
              const ny = Math.cos(phi);
              const nz = Math.sin(phi) * Math.sin(theta);

              const x = radio * nx;
              const y = radio * ny;
              const z = radio * nz;

              vertices.push(x, y, z);       // posición
              vertices.push(nx, ny, nz);    // normal (esfera centrada en origen)
              vertices.push(u, 1.0 - v);    // UV
          }
      }

      const columnas = segmentosLongitud + 1;

      for (let latitud = 0; latitud < segmentosLatitud; latitud++) {
          for (let longitud = 0; longitud < segmentosLongitud; longitud++) {
              const actual = latitud * columnas + longitud;
              const siguiente = actual + columnas;

              indices.push(actual, siguiente, actual + 1);
              indices.push(siguiente, siguiente + 1, actual + 1);
          }
      }

      return {
          vertices: new Float32Array(vertices),
          indices: new Uint16Array(indices)
      };
  }

  COSTO GEOMÉTRICO:
    Latitud | Longitud | Triángulos
    --------+----------+-----------
       8    |    12    |    192
      16    |    24    |    768
      32    |    48    |   3072
      64    |    96    |  12288

ÓRBITA CIRCULAR (PASO 14)
--------------------------------------------------------------------------------
  function crearOrbita(radio, segmentos = 128) {
      const puntos = [];
      for (let i = 0; i < segmentos; i++) {
          const t = i / segmentos;
          const angulo = t * Math.PI * 2.0;
          puntos.push(
              radio * Math.cos(angulo),  // x
              0.0,                       // y
              radio * Math.sin(angulo)   // z
          );
      }
      return new Float32Array(puntos);
  }
  Se dibuja con gl.LINE_LOOP.

ILUMINACIÓN DIFUSA DE LAMBERT (PASO 8)
--------------------------------------------------------------------------------
  FÓRMULA:
    I = max(0, N · L)

  N ... normal normalizada
  L ... dirección hacia la luz normalizada
  · ... producto punto

COMPONENTES
--------------------------------------------------------------------------------
  Ambient ..... Evita zonas totalmente negras (0.08 – 0.20)
  Diffuse ..... Depende de N·L
  Color base .. Color propio del cuerpo

LUZ DIRECCIONAL VS PUNTUAL
--------------------------------------------------------------------------------
  Tipo          | Dato             | Comportamiento
  --------------+------------------+--------------------------------
  Direccional   | uLightDirection  | Rayos paralelos
  Puntual       | uLightPosition   | Dirección cambia por fragmento

  LUZ PUNTUAL (Paso 12):
    vec3 L = normalize(uLightPosition - vWorldPosition);
    float difusa = max(dot(N, L), 0.0);

TRANSFORMACIÓN DE NORMALES
--------------------------------------------------------------------------------
  Simplificación usada: mat3(uModelMatrix) * aNormal
  Correcta para rotaciones y escala uniforme.
  Para escala no uniforme se necesita:
      NormalMatrix = transpose(inverse(mat3(Model)))

CUERPO EMISIVO
--------------------------------------------------------------------------------
  - El Sol usa uEmissive = true → muestra textura sin iluminación.
  - Los planetas usan uEmissive = false → textura × iluminación.

TEXTURA PROCEDURAL (PASO 9)

  function crearTexturaProcedural(gl) {
      const ancho = 8, alto = 4;
      const datos = new Uint8Array(ancho * alto * 4);

      for (let y = 0; y < alto; y++) {
          for (let x = 0; x < ancho; x++) {
              const i = (y * ancho + x) * 4;
              const bloque = (x + y) % 2;
              if (bloque === 0) {
                  datos[i]   = 255; datos[i+1] = 150;
                  datos[i+2] = 20;  datos[i+3] = 255;
              } else {
                  datos[i]   = 20;  datos[i+1] = 70;
                  datos[i+2] = 180; datos[i+3] = 255;
              }
          }
      }

      const textura = gl.createTexture();
      gl.bindTexture(gl.TEXTURE_2D, textura);
      gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, ancho, alto, 0,
                    gl.RGBA, gl.UNSIGNED_BYTE, datos);
      gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
      gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
      gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.REPEAT);
      gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
      return textura;
  }

  function cargarTextura(url) {
      const textura = gl.createTexture();
      gl.bindTexture(gl.TEXTURE_2D, textura);

      // Píxel temporal blanco
      gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, 1, 1, 0, gl.RGBA,
                    gl.UNSIGNED_BYTE, new Uint8Array([255, 255, 255, 255]));

      const imagen = new Image();
      imagen.onload = () => {
          gl.bindTexture(gl.TEXTURE_2D, textura);
          gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, true);
          gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA,
                        gl.UNSIGNED_BYTE, imagen);
          gl.generateMipmap(gl.TEXTURE_2D);
          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER,
                           gl.LINEAR_MIPMAP_LINEAR);
          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.REPEAT);
          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
      };
      imagen.src = url;
      return textura;
  }

TEXTURA DE COLOR SÓLIDO (PASO 13)
--------------------------------------------------------------------------------
  function crearTexturaColor(r, g, b, a = 255) {
      const textura = gl.createTexture();
      gl.bindTexture(gl.TEXTURE_2D, textura);
      gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, 1, 1, 0, gl.RGBA,
                    gl.UNSIGNED_BYTE, new Uint8Array([r, g, b, a]));
      gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
      gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
      return textura;
  }

PARTE XI — ÓRBITAS Y JERARQUÍA

ROTACIÓN VS REVOLUCIÓN
--------------------------------------------------------------------------------
  Movimiento    | Qué cambia                          | Ejemplo
  --------------+-------------------------------------+-------------------
  Rotación      | Orientación del objeto              | Tierra girando
  Revolución    | Posición alrededor de otro punto    | Tierra orbitando

MATRIZ DE ÓRBITA (PASO 11)
--------------------------------------------------------------------------------
  Mórbita = RY(θórbita) · T(distancia, 0, 0)

  COMPOSICIÓN COMPLETA DE LA TIERRA:
    MTierra = Rórbita · Tdistancia · Rpropia · S

  LECTURA DE DERECHA A IZQUIERDA:
    1. S .......... escala
    2. Rpropia .... gira sobre su eje
    3. Tdistancia . coloca a distancia orbital
    4. Rórbita .... gira todo el sistema alrededor del origen

JERARQUÍA SOL → TIERRA → LUNA (PASO 13)
--------------------------------------------------------------------------------
  Matriz del sistema Tierra:
    const tierraSistema = multiplicarMat4(
        rotacionY(orbitaTierra),
        traslacion(3.5, 0.0, 0.0)
    );

  Tierra visual:
    const tierraVisualLocal = multiplicarMat4(
        rotacionY(giroTierra),
        escala(0.55, 0.55, 0.55)
    );
    const modelTierra = multiplicarMat4(tierraSistema, tierraVisualLocal);

  Luna (hereda tierraSistema, NO modelTierra):
    const lunaOrbitaLocal = multiplicarMat4(
        rotacionZ(inclinacionLuna),
        multiplicarMat4(rotacionY(orbitaLuna), traslacion(0.95, 0.0, 0.0))
    );
    const lunaSistemaMundo = multiplicarMat4(tierraSistema, lunaOrbitaLocal);

    const lunaVisualLocal = multiplicarMat4(
        rotacionY(giroLuna),
        escala(0.18, 0.18, 0.18)
    );
    const modelLuna = multiplicarMat4(lunaSistemaMundo, lunaVisualLocal);

  REGLA CLAVE:
    La Luna hereda la TRANSFORMACIÓN DE SISTEMA (tierraSistema),
    no la ESCALA VISUAL (modelTierra).
    Si heredara la escala, la órbita lunar se deformaría.

PARTE XII — RENDER LOOP Y DELTATIME

ESTRUCTURA BÁSICA (PASO 4)
--------------------------------------------------------------------------------
  let angulo = 0.0;
  let tiempoAnterior = 0.0;
  const velocidadAngular = 60 * Math.PI / 180; // rad/s

  function render(tiempoActual) {
      const tiempoSegundos = tiempoActual * 0.001;
      let deltaTime = tiempoSegundos - tiempoAnterior;
      tiempoAnterior = tiempoSegundos;

      if (deltaTime > 0.1) deltaTime = 0.0;  // evitar salto inicial

      angulo += velocidadAngular * deltaTime;

      // ... construir matriz y dibujar

      requestAnimationFrame(render);
  }
  requestAnimationFrame(render);

FÓRMULA DE INTEGRACIÓN TEMPORAL
--------------------------------------------------------------------------------
  incremento = velocidadAngular × deltaTime

  Por qué NO usar ángulo += 0.01:
    Depende de los FPS del equipo. Con deltaTime la velocidad es
    independiente del hardware.

SEPARACIÓN ACTUALIZAR/RENDERIZAR
--------------------------------------------------------------------------------
  Actualizar .... modificar estado (ángulos, posiciones, velocidades)
  Renderizar .... construir matrices, enviar a GPU, dibujar

REQUESTANIMATIONFRAME
--------------------------------------------------------------------------------
  - No es un while tradicional.
  - Solicita al navegador ejecutar render antes del siguiente refresco.
  - Entrega timestamp en milisegundos.

    function obtenerEyeCamara() {
        const cosElevacion = Math.cos(camera.elevacion);
        const x = camera.target[0] + camera.distancia * cosElevacion
                  * Math.sin(camera.azimut);
        const y = camera.target[1] + camera.distancia
                  * Math.sin(camera.elevacion);
        const z = camera.target[2] + camera.distancia * cosElevacion
                  * Math.cos(camera.azimut);
        return [x, y, z];
    }

    function obtenerViewCamara() {
        const eye = obtenerEyeCamara();
        return lookAt(eye, camera.target, [0.0, 1.0, 0.0]);
    }

PARTE XV — CONCEPTOS TRANSVERSALES
================================================================================

HOMOGÉNEAS
--------------------------------------------------------------------------------
  2D: (x, y) → (x, y, 1) con matrices 3×3
  3D: (x, y, z) → (x, y, z, 1) con matrices 4×4
  Permiten representar traslación como multiplicación matricial.

ORDEN DE TRANSFORMACIONES
--------------------------------------------------------------------------------
  T · R · S ≠ S · R · T
  La multiplicación matricial NO es conmutativa.
  La operación más cercana al punto se aplica primero.

DEPTH TEST (PASO 5)
--------------------------------------------------------------------------------
  gl.enable(gl.DEPTH_TEST);
  gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
  Sin esto, superficies traseras se dibujan encima de delanteras.

VIEWPORT
--------------------------------------------------------------------------------
  gl.viewport(0, 0, canvas.width, canvas.height);

ESTADOS DE WEBGL USADOS
--------------------------------------------------------------------------------
  DEPTH_TEST ......... gl.enable(gl.DEPTH_TEST)
  Limpieza color ..... gl.clear(gl.COLOR_BUFFER_BIT)
  Limpieza prof. ..... gl.clear(gl.DEPTH_BUFFER_BIT)

TIPOS DE DATOS JAVASCRIPT → GPU
--------------------------------------------------------------------------------
  JS            | GPU                | Uso
  --------------+--------------------+-----------------------------
  Float32Array  | gl.FLOAT           | Posiciones, normales, UV, matrices
  Uint16Array   | gl.UNSIGNED_SHORT  | Índices
  Uint8Array    | gl.UNSIGNED_BYTE   | Datos de textura

PARTE XXII — CONVENCIONES Y ORDEN DE CONSTRUCCIÓN

CONVENCIONES QUE TE SALVARÁN
--------------------------------------------------------------------------------
  1. Un archivo = una responsabilidad. Si dudas, divide.
  2. Exportaciones nombradas, no default (excepto shaders como string).
  3. Sin variables globales. Todo pasa por import.
  4. Constantes en config/. Nada de números mágicos dispersos.
  5. Shaders como strings en shaders/*.js, no en HTML.
  6. Una textura = una instancia de Texture. No rellenes uniforms a mano.
  7. deltaTime en segundos siempre. Nunca ms dentro de la lógica.
  8. Jerarquía padre-hijo para: muelle→tablones, molino→aspas,
     palmera→hojas, barco→vela.
  9. Sistemas (world/) no dibujan. Solo actualizan estado.
 10. Objetos (scene/objects/) no actualizan tiempo global. Reciben dt.

ESQUELETO DE main.js
--------------------------------------------------------------------------------
  import { initGL } from './core/GLContext.js';
  import { Renderer } from './core/Renderer.js';
  import { Loop } from './core/Loop.js';
  import { Scene } from './scene/Scene.js';
  import { Camera } from './scene/Camera.js';
  import { DirectionalLight } from './scene/lights/DirectionalLight.js';
  import { AmbientLight } from './scene/lights/AmbientLight.js';
  import { DayNightCycle } from './world/DayNightCycle.js';
  import { WindSystem } from './world/WindSystem.js';
  import { WaveSystem } from './world/WaveSystem.js';
  import { Island } from './scene/objects/Island.js';
  import { Windmill } from './scene/objects/Windmill.js';
  import { Dock } from './scene/objects/Dock.js';
  import { Path } from './scene/objects/Path.js';
  import { Boat } from './scene/objects/Boat.js';
  import { PalmTree } from './scene/objects/PalmTree.js';
  import { Rock } from './scene/objects/Rock.js';
  import { Sea } from './scene/objects/Sea.js';
  import { InputManager } from './input/InputManager.js';
  import { CameraController } from './input/CameraController.js';

  const canvas = document.getElementById('glCanvas');
  const gl = initGL(canvas);
  const renderer = new Renderer(gl, canvas);

  const scene = new Scene();
  const camera = new Camera({ target: [0, 1, 0], distancia: 20 });
  const sun = new DirectionalLight();
  const ambient = new AmbientLight(0.15);
  scene.add(sun, ambient);

  const dayNight = new DayNightCycle(sun);
  const wind = new WindSystem();
  const waves = new WaveSystem();

  const island  = new Island();
  const molino  = new Windmill();
  const muelle  = new Dock();
  const camino  = new Path();
  const barco   = new Boat();
  const palmera = new PalmTree();
  const roca    = new Rock();
  const mar     = new Sea(waves);

  scene.add(island, molino, muelle, camino, barco, palmera, roca, mar);

  const input = new InputManager(canvas);
  const camCtrl = new CameraController(camera, input);

  const loop = new Loop((dt, t) => {
    dayNight.update(dt);
    wind.update(dt);
    waves.update(dt);
    camCtrl.update(dt);

    barco.update(dt, waves);       // flotación
    palmera.update(dt, wind);      // viento
    molino.update(dt, wind);       // aspas giran

    scene.updateWorldMatrices();
    renderer.render(scene, camera, { sun, ambient });
  });

  loop.start();
